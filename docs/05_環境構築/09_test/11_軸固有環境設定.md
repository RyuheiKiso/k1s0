## 演習 1: coverage matrix 80 cell 局所生成演習

95 cell matrix（5 verification_class × 19 軸）の局所版（80 cell 分）を手元で生成する演習。全 cell を一度に解決する必要はないが、生成スクリプトが動くことを確認する。

```python
# /tmp/gen_coverage_matrix.py
VERIFICATION_CLASSES = [
    "unit", "contract", "integration", "chaos", "mutation"
]
AXES = [
    "07_bidi", "08_migration", "09_observability", "10_auth",
    "11_key_mgmt", "12_schema_evolution", "13_slo", "14_oss_lifecycle",
    "15_tenant_capacity", "32_tenant_isolation", "39_client_state",
    "infra01", "data01", "security15", "security16",
    "client15", "ops15", "test15", "formal15"
]

print(f"Total cells: {len(VERIFICATION_CLASSES) * len(AXES)}")
print("\n=== Coverage Matrix (80 cell preview) ===")
count = 0
for axis in AXES:
    for vc in VERIFICATION_CLASSES:
        if count >= 80:
            break
        print(f"  [{vc}] x [{axis}]: NOT_COVERED")
        count += 1
    if count >= 80:
        break
print(f"\nGenerated {count} cell entries")
```

```bash
python3 /tmp/gen_coverage_matrix.py
```

## 演習 2: mutation score 演習（cargo-mutants on tier1 sample）

```bash
# tier1 の Rust コードに対して cargo-mutants を実行
# src/ 配下に Rust project があることを前提とする
ls src/*/Cargo.toml 2>/dev/null || echo "Rust project not found, using /tmp/sample"

# sample Rust project で演習
mkdir -p /tmp/mutation_sample/src
cat > /tmp/mutation_sample/Cargo.toml << 'EOF'
[package]
name = "mutation_sample"
version = "0.1.0"
edition = "2021"
EOF
cat > /tmp/mutation_sample/src/lib.rs << 'EOF'
pub fn add(a: i32, b: i32) -> i32 { a + b }
pub fn is_even(n: i32) -> bool { n % 2 == 0 }
#[cfg(test)]
mod tests {
    use super::*;
    #[test] fn test_add() { assert_eq!(add(1, 2), 3); }
    #[test] fn test_even() { assert!(is_even(4)); assert!(!is_even(3)); }
}
EOF

cd /tmp/mutation_sample
cargo-mutants --list 2>&1 | head -10
cargo-mutants -- --test-threads=1 2>&1 | tail -10
```

## 演習 3: Pact contract test broker 起動 + sample publish

```bash
# Pact broker 起動（07_テスト検証環境 と同じ compose ファイル）
docker compose -f /tmp/pact-compose.yml up -d

# broker の起動待ち
until curl -s -u pact:pact http://localhost:9292/diagnostic/status > /dev/null 2>&1; do
  echo "Waiting for Pact broker..."; sleep 3
done

# sample pact ファイルの publish
cat > /tmp/sample.pact.json << 'EOF'
{
  "consumer": {"name": "SampleConsumer"},
  "provider": {"name": "SampleProvider"},
  "interactions": [
    {
      "description": "a sample request",
      "request": {"method": "GET", "path": "/sample"},
      "response": {"status": 200, "body": {"status": "ok"}}
    }
  ],
  "metadata": {"pactSpecification": {"version": "2.0.0"}}
}
EOF

curl -s -u pact:pact \
  -X PUT \
  -H "Content-Type: application/json" \
  -d @/tmp/sample.pact.json \
  "http://localhost:9292/pacts/provider/SampleProvider/consumer/SampleConsumer/version/1.0.0" \
  && echo "Pact published OK"
```
